After running the container and attaching to it:
    docker exec -it meteornet bash

You need to symlink the libcublas and libcudart files
sudo mkdir -p /usr/local/cuda-9.0/lib64

sudo ln -sf /usr/local/cuda-9.0/targets/x86_64-linux/lib/libcublas.so.9.0.176 \
             /usr/local/cuda-9.0/lib64/libcublas.so

sudo ln -sf /usr/local/cuda-9.0/targets/x86_64-linux/lib/libcublas.so.9.0.176 \
             /usr/local/cuda-9.0/lib64/libcublas.so.9.0

sudo ln -sf /usr/local/cuda-9.0/targets/x86_64-linux/lib/libcublas.so.9.0.176 \
             /usr/local/cuda-9.0/lib64/libcublas.so.9.0.176

sudo ln -sf /usr/local/cuda-9.0/targets/x86_64-linux/lib/libcudart.so.9.0 \
             /usr/local/cuda-9.0/lib64/libcudart.so

sudo ln -sf /usr/local/cuda-9.0/targets/x86_64-linux/lib/libcudart.so.9.0 \
             /usr/local/cuda-9.0/lib64/libcudart.so.9.0

sudo ln -sf /usr/local/cuda-9.0/targets/x86_64-linux/lib/libcuda.so \
             /usr/local/cuda-9.0/lib64/libcuda.so

echo "/usr/local/cuda-9.0/lib64" | sudo tee /etc/ld.so.conf.d/cuda-9-0.conf
sudo ldconfig
